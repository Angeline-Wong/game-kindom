<script>
const edict = document.getElementById("edict");
const edictText = document.getElementById("edictText");
const stampTrigger = document.getElementById("stampTrigger");
const stampMark = document.getElementById("stampMark");
const shock = document.getElementById("shock");
const particles = document.getElementById("particles");
const hint = document.getElementById("hint");
const startBtn = document.getElementById("startBtn");
const complete = document.getElementById("complete");

function spawnParticles(){
  particles.innerHTML = "";
  for(let i=0;i<18;i++){
    const p = document.createElement("span");
    p.className = "particle";
    p.style.left = "66%";
    p.style.top = "73%";
    const angle = Math.random()*Math.PI*2;
    const distance = 35 + Math.random()*60;
    p.style.setProperty("--dx", `${Math.cos(angle)*distance}px`);
    p.style.setProperty("--dy", `${Math.sin(angle)*distance}px`);
    p.style.animation = `particleFly ${0.55+Math.random()*0.35}s ease-out forwards`;
    particles.appendChild(p);
  }
}

startBtn.addEventListener("click",()=>{
  startBtn.classList.add("hidden");
  hint.textContent = "圣旨展开中……";
  edict.classList.add("open");

  setTimeout(()=>{
    edictText.classList.add("show");
    hint.textContent = "诏书内容已显现";
  },1400);

  setTimeout(()=>{
    stampTrigger.classList.add("show");
    hint.textContent = "点击“奉旨用印”完成诏书";
  },2600);
});

stampTrigger.addEventListener("click",()=>{
  if(stampTrigger.classList.contains("pressed")) return;

  stampTrigger.classList.add("pressed");
  hint.textContent = "御印落下！";

  setTimeout(()=>{
    stampMark.classList.add("stamped");
    shock.classList.add("flash");
    spawnParticles();
    if(navigator.vibrate) navigator.vibrate(35);

    document.querySelector(".stage").animate(
      [
        {transform:"translate(0,0)"},
        {transform:"translate(-2px,2px)"},
        {transform:"translate(2px,-1px)"},
        {transform:"translate(0,0)"}
      ],
      {duration:260,easing:"ease-out"}
    );
  },190);

  setTimeout(()=>{
    hint.textContent = "圣旨已加盖御印";
  },580);

  setTimeout(()=>{
    complete.classList.add("show");
  },1120);
});
</script>
