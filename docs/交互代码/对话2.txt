<script>
const choices=[
  {text:'口误',speaker:'你',reply:'朕方才一时口误，母后莫要见怪。'},
  {text:'坦白',speaker:'你',reply:'朕确有一事，想向母后如实说明。'},
  {text:'沉默',speaker:'太后',reply:'你既不愿开口，哀家便不再逼问。'},
  {text:'转移话题',speaker:'你',reply:'今日御花园景色正好，不如陪朕走走。'}
].slice(0,4);
const panel=document.getElementById('choicePanel');
const speaker=document.getElementById('speaker');
const text=document.getElementById('dialogText');
function renderChoices(){
  panel.innerHTML='';
  choices.forEach((item,index)=>{
    const button=document.createElement('button');
    button.className='choice-btn';button.textContent=item.text;
    button.onclick=()=>{
      speaker.textContent=item.speaker;text.textContent=item.reply;
      [...panel.children].forEach((el,i)=>{el.disabled=true;el.classList.toggle('selected',i===index)});
    };
    panel.appendChild(button);
  });
}
renderChoices();
document.getElementById('nextBtn').onclick=()=>panel.classList.toggle('hidden');
document.getElementById('skipBtn').onclick=()=>{
  document.getElementById('gameUi').animate([{opacity:1},{opacity:0}],{duration:260,fill:'forwards'});
};
</script>
