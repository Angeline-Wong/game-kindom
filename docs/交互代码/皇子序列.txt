let currentType = "全部";
let currentSort = "order";
let currentDirection = "desc";
let selected = null;

const cards = document.getElementById("cards");
const searchInput = document.getElementById("searchInput");
const detailSheet = document.getElementById("detailSheet");
const sortSheet = document.getElementById("sortSheet");
const sortTrigger = document.getElementById("sortTrigger");
const toast = document.getElementById("toast");

function sortLabel(){
  const map = {
    order:"序位",
    age:"年龄",
    favor:"宠爱",
    talent:"资质",
    mother:"生母",
    type:"身份"
  };
  return map[currentSort];
}

function compare(a,b){
  let av;
  let bv;

  if(currentSort === "type"){
    av = typeWeight[a.type] || 0;
    bv = typeWeight[b.type] || 0;
  }else{
    av = a[currentSort];
    bv = b[currentSort];
  }

  if(typeof av === "string"){
    return currentDirection === "asc"
      ? av.localeCompare(bv,"zh-CN")
      : bv.localeCompare(av,"zh-CN");
  }

  return currentDirection === "asc" ? av-bv : bv-av;
}

function matchType(item){
  if(currentType === "全部") return true;
  if(currentType === "幼年") return item.age < 12;
  if(currentType === "成年") return item.age >= 16;
  return item.type === currentType;
}

function getData(){
  const keyword = searchInput.value.trim();

  return heirs
    .filter(matchType)
    .filter(item => !keyword || item.name.includes(keyword) || item.mother.includes(keyword))
    .sort(compare);
}

function render(){
  const data = getData();
  document.getElementById("totalCount").textContent = data.length;
  sortTrigger.textContent = `${sortLabel()} ${currentDirection === "asc" ? "↑" : "↓"}`;

  cards.innerHTML = "";

  data.forEach((item,index)=>{
    const card = document.createElement("article");
    card.className = "card" + (selected?.name === item.name ? " selected" : "");
    card.style.animationDelay = `${index * 45}ms`;

    card.innerHTML = `
      <div class="portrait" data-initial="${item.name.slice(-1)}">
        <span class="quality">${item.age >= 16 ? "成年" : "幼年"}</span>
      </div>
      <span class="rank-ribbon">${item.title}</span>

      <div class="name-row">
        <span class="name">${item.name}</span>
        <span class="mother">生母：${item.mother}</span>
      </div>

      <div class="line"></div>

      <div class="stats">
        <span><b>${item.age}</b>岁</span>
        <span><b>${item.favor}</b>宠爱</span>
        <span><b>${item.talent}</b>资质</span>
      </div>

      <div class="status">
        ${item.status.map((status,i)=>`
          <span class="badge ${i === 0 && (status.includes("储君") || status.includes("可册封")) ? "hot" : ""}">
            ${status}
          </span>
        `).join("")}
      </div>
    `;

    card.onclick = ()=>{
      selected = item;
      render();

      document.getElementById("sheetTitle").textContent = `${item.name} · ${item.title}`;
      document.getElementById("sheetSub").textContent =
        `生母${item.mother} · ${item.age}岁 · 文才${item.literary} · 武艺${item.martial}`;

      sortSheet.classList.remove("show");
      detailSheet.classList.add("show");
    };

    cards.appendChild(card);
  });
}

function showToast(text){
  toast.textContent = text;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer = setTimeout(()=>toast.classList.remove("show"),1100);
}

  button.onclick = ()=>{
    document.querySelectorAll(".tab").forEach(item=>item.classList.remove("active"));
    button.classList.add("active");
    currentType = button.dataset.type;
    detailSheet.classList.remove("show");
    render();
  };
});

  detailSheet.classList.remove("show");
  render();
};

  detailSheet.classList.remove("show");
  sortSheet.classList.toggle("show");
};

  button.onclick = ()=>{
    currentSort = button.dataset.sort;

    document.querySelectorAll(".sort-option").forEach(item=>{
      item.classList.remove("active");
      item.lastElementChild.textContent = "";
    });

    button.classList.add("active");
    button.lastElementChild.textContent = "✓";
    render();
  };
});

  button.onclick = ()=>{
    currentDirection = button.dataset.direction;

    document.querySelectorAll("[data-direction]").forEach(item=>{
      item.classList.remove("active");
    });

    button.classList.add("active");
    render();
  };
});

    if(selected){

    }
  };
});
