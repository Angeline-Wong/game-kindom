let filter="全部";
let vacancyOnly=true;
let selected=null;

const list=document.getElementById("officeList");
const confirmBtn=document.getElementById("confirmBtn");
const searchInput=document.getElementById("searchInput");
const toast=document.getElementById("toast");

function canTake(office){
  const rankAllowed = SETTINGS.allowLowerRank
    ? office.rankLevel >= MINISTER.rankLevel && office.rankLevel <= MINISTER.rankLevel + 1
    : office.rankLevel === MINISTER.rankLevel;
  const skillValue = MINISTER.skills[office.skill] || 0;
  return rankAllowed && skillValue >= office.need && office.vacant;
}

function render(){
  const keyword=searchInput.value.trim();
  const rows=offices.filter(o=>{
    const category=filter==="全部" || o.area===filter || o.type===filter;
    const vacancy=!vacancyOnly || o.vacant;
    const search=!keyword || o.name.includes(keyword) || o.agency.includes(keyword);
    return category && vacancy && search;
  });

  list.innerHTML="";
  if(!rows.length){
    list.innerHTML='<div class="empty">暂无符合条件的官职</div>';
    return;
  }

  rows.forEach(o=>{
    const eligible=canTake(o);
    const skillValue=MINISTER.skills[o.skill] || 0;
    const rankMismatch=o.rankLevel!==MINISTER.rankLevel;
    const reason=!o.vacant?"暂无空缺":rankMismatch?`需${o.rank}`:skillValue<o.need?`${o.skill}需${o.need}`:"可调任";

    const card=document.createElement("article");
    card.className=`office-card ${eligible?"":"disabled"} ${selected===o?"selected":""}`;
    card.innerHTML=`
      <span class="check">✓</span>
      <div class="office-icon">${o.agency.slice(0,1)}</div>
      <div>
        <div class="office-name">${o.name}</div>
        <div class="office-meta">${o.agency} · ${o.area} · ${o.type} · ${o.rank}</div>
        <div class="office-desc">${o.duty}</div>
        <div class="tags">
          <span class="tag">${o.skill} ${skillValue}/${o.need}</span>
          <span class="tag">${o.vacant?"有空缺":"已任满"}</span>
        </div>
      </div>
      <div class="office-side">
        <b>${reason}</b>
        权势 ${signed(o.power)}<br>
        俸禄 ${signed(o.salary)}
      </div>
    `;
    card.onclick=()=>{
      if(!eligible){
        showToast(reason);
        return;
      }
      selected=o;
      confirmBtn.disabled=false;
      render();
    };
    list.appendChild(card);
  });
}

function signed(value){
  if(value===0)return "不变";
  return value>0?`+${value}`:`${value}`;
}

document.querySelectorAll(".tab").forEach(btn=>{
  btn.onclick=()=>{
    document.querySelectorAll(".tab").forEach(x=>x.classList.remove("active"));
    btn.classList.add("active");
    filter=btn.dataset.filter;
    selected=null;
    confirmBtn.disabled=true;
    render();
  };
});

document.getElementById("vacancyBtn").onclick=e=>{
  vacancyOnly=!vacancyOnly;
  e.currentTarget.classList.toggle("active",vacancyOnly);
  e.currentTarget.textContent=vacancyOnly?"仅看空缺":"显示全部";
  selected=null;
  confirmBtn.disabled=true;
  render();
};

searchInput.oninput=render;

confirmBtn.onclick=()=>{

  document.getElementById("targetOffice").innerHTML=

  document.getElementById("powerEffect").textContent=signed(selected.power);
  document.getElementById("salaryEffect").textContent=signed(selected.salary);
  document.getElementById("fitEffect").textContent=selected.fit;
  document.getElementById("confirmLayer").classList.add("show");
};

document.getElementById("backBtn").onclick=()=>document.getElementById("confirmLayer").classList.remove("show");
document.getElementById("executeBtn").onclick=()=>{
  document.getElementById("confirmLayer").classList.remove("show");
  showToast(`已颁旨：调任张居正为${selected.name}`);
};
document.getElementById("cancelBtn").onclick=()=>showToast("已取消调任");
document.getElementById("closeBtn").onclick=()=>document.getElementById("panel").style.display="none";

function showToast(text){
  toast.textContent=text;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer=setTimeout(()=>toast.classList.remove("show"),1300);
}
