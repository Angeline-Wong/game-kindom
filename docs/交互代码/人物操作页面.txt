const interaction = document.getElementById("interaction");
const toast = document.getElementById("toast");
const visibleCount = 6;

function showToast(name){
  toast.textContent = "已选择：" + name;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer = setTimeout(()=>toast.classList.remove("show"),1100);
}

function render(){
  interaction.innerHTML = '<div class="path"></div>';
  const list = actions.slice(0,visibleCount);
  const usableHeight = 380;
  const step = list.length > 1 ? usableHeight/(list.length-1) : 0;

  list.forEach((action,index)=>{
    const button = document.createElement("button");
    button.className = "item";

    // 轻微弧线排布，贴近参考图
    const curveOffset = Math.sin((index/(list.length-1))*Math.PI) * 13;
    button.style.top = `${index*step}px`;
    button.style.left = `${curveOffset}px`;

    button.innerHTML = `
      <span class="icon">${action.icon}</span>
      <span class="tag">${action.name}</span>
    `;

    button.addEventListener("click",()=>{
      document.querySelectorAll(".item").forEach(el=>el.classList.remove("selected"));
      button.classList.add("selected");
      showToast(action.name);
    });

    interaction.appendChild(button);
  });
}

document.getElementById("backBtn").addEventListener("click",()=>showToast("返回"));
render();
