    {name:"林月华",rank:"嫔",palace:"永寿宫"}
  ],
  [
    {name:"贺兰雪",rank:"妃",palace:"承乾宫"},
    {name:"许知意",rank:"贵人",palace:"钟粹宫"},
    {name:"姜晚棠",rank:"嫔",palace:"永和宫"},
    {name:"顾明珠",rank:"皇后",palace:"坤宁宫"},
    {name:"秦若仪",rank:"常在",palace:"景阳宫"},
    {name:"赵明珮",rank:"答应",palace:"储秀宫"},
    {name:"温婉柔",rank:"贵人",palace:"延禧宫"},
    {name:"谢清歌",rank:"嫔",palace:"咸福宫"}
  ]
];

let currentPage = 0;
let selected = null;

const rack = document.getElementById("rack");
const pageInfo = document.getElementById("pageInfo");
const confirmBtn = document.getElementById("confirmBtn");
const confirmLayer = document.getElementById("confirmLayer");
const toast = document.getElementById("toast");

function render(direction="none"){
  if(direction !== "none"){
    rack.style.animation = direction === "next"
      ? "pageOutLeft .18s ease forwards"
      : "pageOutRight .18s ease forwards";
  }

  setTimeout(()=>{
    rack.innerHTML = "";

    pages[currentPage].forEach(item=>{
      const slot = document.createElement("div");
      slot.className = "牌-slot";

      const button = document.createElement("button");
      button.className = "green牌";
      button.innerHTML = `
        <span class="牌-head"></span>
        <span class="牌-body">
          <span class="牌-text">
            <span class="牌-name">${item.name}</span>
            <span class="牌-rank">${item.rank}</span>
          </span>
          <span class="牌-palace">${item.palace}</span>
        </span>
        <span class="selected-check">✓</span>
      `;

      button.addEventListener("click",()=>{
        document.querySelectorAll(".green牌").forEach(el=>el.classList.remove("selected"));
        button.classList.add("selected");
        selected = item;
        confirmBtn.disabled = false;
      });

      slot.appendChild(button);
      rack.appendChild(slot);
    });

    pageInfo.textContent = `第 ${currentPage + 1} / ${pages.length} 盘`;
    document.getElementById("prevBtn").disabled = currentPage === 0;
    document.getElementById("nextBtn").disabled = currentPage === pages.length - 1;

    if(direction !== "none"){
      rack.style.animation = direction === "next"
        ? "pageInRight .24s ease forwards"
        : "pageInLeft .24s ease forwards";
    }else{
      rack.style.animation = "";
    }
  }, direction === "none" ? 0 : 180);
}

function showToast(text){
  toast.textContent = text;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer = setTimeout(()=>toast.classList.remove("show"),1100);
}

document.getElementById("prevBtn").addEventListener("click",()=>{
  if(currentPage === 0) return;
  currentPage--;
  selected = null;
  confirmBtn.disabled = true;
  render("prev");
});

document.getElementById("nextBtn").addEventListener("click",()=>{
  if(currentPage === pages.length - 1) return;
  currentPage++;
  selected = null;
  confirmBtn.disabled = true;
  render("next");
});

confirmBtn.addEventListener("click",()=>{
  if(!selected) return;

  document.getElementById("confirm牌").textContent = selected.name;
  document.getElementById("confirmName").textContent = `${selected.name} · ${selected.rank}`;
  document.getElementById("confirmText").textContent =
    `今夜传召${selected.palace}${selected.name}侍寝？`;

  confirmLayer.classList.add("show");
});

document.getElementById("cancelBtn").addEventListener("click",()=>{
  confirmLayer.classList.remove("show");
});

document.getElementById("summonBtn").addEventListener("click",()=>{
  confirmLayer.classList.remove("show");
  showToast(`已传旨：今夜临幸${selected.name}`);
});

document.getElementById("closeBtn").addEventListener("click",()=>{
  document.getElementById("panel").style.display = "none";
});

render();
</script>
