let currentType = "全部";
let currentYear = "全部年份";

const timeline = document.getElementById("timeline");
const searchInput = document.getElementById("searchInput");
const detailSheet = document.getElementById("detailSheet");
const toast = document.getElementById("toast");

function render(){
  const keyword = searchInput.value.trim();
  timeline.innerHTML = "";

  annals.forEach(group=>{
    if(currentYear !== "全部年份" && group.year !== currentYear) return;

    const filtered = group.events.filter(event=>{
      const typeMatch = currentType === "全部" || event.type === currentType;
      const keywordMatch = !keyword || event.title.includes(keyword) || event.brief.includes(keyword);
      return typeMatch && keywordMatch;
    });

    if(!filtered.length) return;

    const wrap = document.createElement("section");
    wrap.className = "year-group";
    wrap.innerHTML = `
      <span class="year-marker"></span>
      <h3 class="year-title">${group.year}</h3>
      <div class="year-note">${group.note}</div>
    `;

    filtered.forEach(event=>{
      const item = document.createElement("article");
      item.className = `event ${event.tone}`;
      item.innerHTML = `
        <div class="event-head">
          <span class="event-title">${event.title}</span>
          <span class="event-tag">${event.type}</span>
        </div>
        <p>${event.brief}</p>
      `;

      item.onclick = ()=>{
        document.getElementById("sheetTitle").textContent = event.title;
        document.getElementById("sheetSub").textContent = `${group.year} · ${event.type}`;
        document.getElementById("sheetBody").textContent = event.detail;
        detailSheet.classList.add("show");
      };

      wrap.appendChild(item);
    });

    timeline.appendChild(wrap);
  });

  if(!timeline.children.length){
    timeline.innerHTML = `<div style="text-align:center;color:#9f8555;padding:35px 0">暂无符合条件的史录</div>`;
  }
}

document.querySelectorAll(".tab").forEach(button=>{
  button.onclick = ()=>{
    document.querySelectorAll(".tab").forEach(item=>item.classList.remove("active"));
    button.classList.add("active");
    currentType = button.dataset.type;
    detailSheet.classList.remove("show");
    render();
  };
});

searchInput.oninput = ()=>{
  detailSheet.classList.remove("show");
  render();
};

document.getElementById("yearBtn").onclick = ()=>{
  const years = ["全部年份","永和三年","永和二年","永和元年"];
  const currentIndex = years.indexOf(currentYear);
  currentYear = years[(currentIndex + 1) % years.length];
  document.getElementById("yearBtn").textContent = currentYear;
  render();
};

document.getElementById("sheetClose").onclick = ()=>{
  detailSheet.classList.remove("show");
};

document.getElementById("closeBtn").onclick = ()=>{
  document.getElementById("panel").style.display = "none";
  detailSheet.classList.remove("show");
};

render();
</script>
