<script>
  const lines = [
    { speaker: '你', text: '皇后，瞧你看什么呢？出神了。' },
    { speaker: '皇后', text: '臣妾正在想，御花园的海棠该开了。' },
    { speaker: '你', text: '那便陪朕去走走。' }
  ];

  let index = 0;
  const speaker = document.getElementById('speaker');
  const dialogText = document.getElementById('dialogText');
  const hint = document.getElementById('hint');

  document.getElementById('nextBtn').addEventListener('click', () => {
    index = (index + 1) % lines.length;
    speaker.textContent = lines[index].speaker;
    dialogText.animate(
      [{ opacity: 0, transform: 'translateY(4px)' }, { opacity: 1, transform: 'translateY(0)' }],
      { duration: 220, easing: 'ease-out' }
    );
    dialogText.textContent = lines[index].text;
    hint.textContent = index === lines.length - 1 ? '再次点击将从第一句重新播放' : '点击右下角金色箭头查看下一句';
  });

  document.getElementById('skipBtn').addEventListener('click', () => {
    document.querySelector('.dialog-wrap').animate(
      [{ opacity: 1, transform: 'translateY(0)' }, { opacity: 0, transform: 'translateY(10px)' }],
      { duration: 260, fill: 'forwards', easing: 'ease-in' }
    );
    hint.textContent = '对话已跳过';
  });
</script>
