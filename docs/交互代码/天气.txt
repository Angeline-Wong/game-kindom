<script>
const bgImage = document.getElementById("backgroundImage");
const emptyState = document.getElementById("emptyState");
const fileInput = document.getElementById("fileInput");
const fileName = document.getElementById("fileName");
const weatherLayer = document.getElementById("weatherLayer");
const controls = document.getElementById("controls");
const fitBtn = document.getElementById("fitBtn");

let thunderTimer = null;
let fitMode = "cover";

function clearWeather(){
  weatherLayer.innerHTML = "";
  clearInterval(thunderTimer);
}

function createRain(){
  for(let i=0;i<72;i++){
    const drop=document.createElement("span");
    drop.className="rain-drop";
    drop.style.left=`${Math.random()*115}%`;
    drop.style.animationDuration=`${.58+Math.random()*.68}s`;
    drop.style.animationDelay=`-${Math.random()*2}s`;
    drop.style.opacity=`${.26+Math.random()*.48}`;
    weatherLayer.appendChild(drop);
  }
}

function createSnow(){
  const symbols=["•","✦","❄"];
  for(let i=0;i<56;i++){
    const flake=document.createElement("span");
    flake.className="snowflake";
    flake.textContent=symbols[Math.floor(Math.random()*symbols.length)];
    flake.style.left=`${Math.random()*100}%`;
    flake.style.fontSize=`${7+Math.random()*13}px`;
    flake.style.animationDuration=`${5+Math.random()*7}s`;
    flake.style.animationDelay=`-${Math.random()*9}s`;
    flake.style.setProperty("--sway",`${-38+Math.random()*76}px`);
    flake.style.opacity=`${.42+Math.random()*.54}`;
    weatherLayer.appendChild(flake);
  }
}

function createSunny(){
  for(let i=0;i<28;i++){
    const dust=document.createElement("span");
    dust.className="dust";
    dust.style.left=`${8+Math.random()*84}%`;
    dust.style.top=`${12+Math.random()*72}%`;
    dust.style.animationDuration=`${3+Math.random()*4}s`;
    dust.style.animationDelay=`-${Math.random()*4}s`;
    weatherLayer.appendChild(dust);
  }
}

function createNight(){
  for(let i=0;i<30;i++){
    const star=document.createElement("span");
    star.className="star";
    star.style.left=`${Math.random()*100}%`;
    star.style.top=`${Math.random()*62}%`;
    star.style.animationDuration=`${1.4+Math.random()*3}s`;
    star.style.animationDelay=`-${Math.random()*3}s`;
    weatherLayer.appendChild(star);
  }
}

function createThunder(){
  createRain();

  const bolt=document.createElement("span");
  bolt.className="lightning";
  bolt.innerHTML=`<svg viewBox="0 0 100 300">
    <polyline points="56,0 40,68 61,68 34,139 54,139 22,224 49,159 29,159 59,92 39,92 68,0"/>
  </svg>`;
  weatherLayer.appendChild(bolt);

  const flash=()=>{
    weatherLayer.classList.remove("flash");
    void weatherLayer.offsetWidth;
    weatherLayer.classList.add("flash");
  };

  flash();
  thunderTimer=setInterval(flash,4600);
}

function setWeather(type){
  clearWeather();
  weatherLayer.className=`weather-overlay weather-${type}`;

  if(type==="rain") createRain();
  if(type==="snow") createSnow();
  if(type==="sunny") createSunny();
  if(type==="night") createNight();
  if(type==="thunder") createThunder();

  document.querySelectorAll("[data-weather]").forEach(btn=>{
    btn.classList.toggle("active",btn.dataset.weather===type);
  });
}

fileInput.addEventListener("change",()=>{
  const file=fileInput.files && fileInput.files[0];
  if(!file) return;

  const reader=new FileReader();

  reader.onload=event=>{
    bgImage.onload=()=>{
      bgImage.classList.add("show");
      emptyState.classList.add("hidden");
    };

    bgImage.onerror=()=>{
      alert("背景图读取失败，请换一张 JPG 或 PNG 图片。");
    };

    bgImage.src=event.target.result;
    fileName.textContent=file.name;
  };

  reader.readAsDataURL(file);
});

fitBtn.addEventListener("click",()=>{
  fitMode = fitMode === "cover" ? "contain" : "cover";
  bgImage.style.objectFit = fitMode;
  fitBtn.textContent = fitMode === "cover" ? "填满" : "完整";
});

document.querySelectorAll("[data-weather]").forEach(btn=>{
  btn.addEventListener("click",()=>setWeather(btn.dataset.weather));
});

document.getElementById("toggleControls").addEventListener("click",event=>{
  controls.classList.toggle("hidden");
  event.currentTarget.textContent = controls.classList.contains("hidden") ? "显示" : "隐藏";
});

setWeather("rain");
</script>
