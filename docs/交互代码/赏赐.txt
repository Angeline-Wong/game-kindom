<script>
let selectedType = null;
const selectedTreasures = new Map();

const rewardCards = document.querySelectorAll(".reward-card");
const confirmBtn = document.getElementById("confirmBtn");
const treasuryLayer = document.getElementById("treasuryLayer");
const treasureValue = document.getElementById("treasureValue");
const chooseTreasureBtn = document.getElementById("chooseTreasureBtn");
const toast = document.getElementById("toast");
const selectedKinds = document.getElementById("selectedKinds");
const selectedCount = document.getElementById("selectedCount");

function showToast(text){
  toast.textContent = text;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer = setTimeout(()=>toast.classList.remove("show"),1200);
}

rewardCards.forEach(card=>{
  card.addEventListener("click",()=>{
    rewardCards.forEach(item=>item.classList.remove("selected"));
    card.classList.add("selected");
    selectedType = card.dataset.type;

    if(selectedType === "treasure"){
      treasuryLayer.classList.add("show");
      confirmBtn.disabled = selectedTreasures.size === 0;
    }else{
      confirmBtn.disabled = false;
    }
  });
});

function updateTreasureSummary(){
  let kinds = 0;
  let total = 0;

  document.querySelectorAll(".treasure").forEach(item=>{
    const name = item.dataset.name;
    const count = selectedTreasures.get(name) || 0;
    const stock = Number(item.dataset.stock);

    item.querySelector(".qty-value").textContent = count;
    item.querySelector(".minus").disabled = count <= 0;
    item.querySelector(".plus").disabled = count >= stock;
    item.classList.toggle("selected", count > 0);

    if(count > 0){
      kinds++;
      total += count;
    }
  });

  selectedKinds.textContent = kinds;
  selectedCount.textContent = total;
  chooseTreasureBtn.disabled = total === 0;
}

document.querySelectorAll(".treasure").forEach(item=>{
  const name = item.dataset.name;
  const stock = Number(item.dataset.stock);

  item.querySelector(".plus").addEventListener("click",()=>{
    const current = selectedTreasures.get(name) || 0;
    if(current >= stock) return;
    selectedTreasures.set(name,current + 1);
    updateTreasureSummary();
  });

  item.querySelector(".minus").addEventListener("click",()=>{
    const current = selectedTreasures.get(name) || 0;
    if(current <= 1){
      selectedTreasures.delete(name);
    }else{
      selectedTreasures.set(name,current - 1);
    }
    updateTreasureSummary();
  });
});

chooseTreasureBtn.addEventListener("click",()=>{
  if(selectedTreasures.size === 0) return;

  const summary = [...selectedTreasures.entries()]
    .map(([name,count])=>`${name}×${count}`)
    .join("、");

  treasureValue.textContent = summary;
  treasuryLayer.classList.remove("show");
  confirmBtn.disabled = false;
});

document.getElementById("treasuryClose").addEventListener("click",()=>{
  treasuryLayer.classList.remove("show");
});

document.getElementById("backBtn").addEventListener("click",()=>{
  treasuryLayer.classList.remove("show");
});

confirmBtn.addEventListener("click",()=>{
  const labels = {
    ink:"御赐墨宝",
    silver:"赏银百两",
    gold:"赏金百两",
    treasure:[...selectedTreasures.entries()]
      .map(([name,count])=>`${name}×${count}`)
      .join("、") || "国库珍宝"
  };

  if(!selectedType) return;
  showToast(`已赏赐：${labels[selectedType]}`);
});

document.getElementById("cancelBtn").addEventListener("click",()=>{
  showToast("已取消赏赐");
});

updateTreasureSummary();

document.getElementById("closeBtn").addEventListener("click",()=>{
  document.getElementById("rewardPanel").style.display = "none";
});
