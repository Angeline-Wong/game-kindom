<script>
  const dialogueData = {
    speaker:'太后',
    text:'皇帝，你说什么？',
    choices:[
      {text:'口误',speaker:'你',reply:'儿臣一时口误，还请母后莫怪。'},
      {text:'坦白',speaker:'你',reply:'儿臣确有一事，想向母后如实禀明。'},
      {text:'沉默',speaker:'你',reply:'儿臣只是……一时不知该如何开口。'},
      {text:'转移话题',speaker:'你',reply:'母后，近日天气转凉，还请多加保重。'}
    ]
  };

  const speakerEl=document.getElementById('speaker');
  const textEl=document.getElementById('dialogueText');
  const panelEl=document.getElementById('choicePanel');
  const nextBtn=document.getElementById('nextBtn');
  const skipBtn=document.getElementById('skipBtn');
  const gameUi=document.getElementById('gameUi');

  function renderChoices(choices){
    panelEl.innerHTML='';
    choices.slice(0,4).forEach(choice=>{
      const btn=document.createElement('button');
      btn.className='choice-btn';
      btn.textContent=choice.text;
      btn.addEventListener('click',()=>selectChoice(choice,btn));
      panelEl.appendChild(btn);
    });
  }

  function selectChoice(choice,selectedBtn){
    panelEl.querySelectorAll('.choice-btn').forEach(btn=>{
      btn.disabled=true;
      if(btn===selectedBtn)btn.classList.add('selected');
    });
    speakerEl.textContent=choice.speaker;
    textEl.textContent=choice.reply;
    textEl.animate([
      {opacity:0,transform:'translateY(4px)'},
      {opacity:1,transform:'translateY(0)'}
    ],{duration:220,easing:'ease-out'});
  }

  nextBtn.addEventListener('click',()=>panelEl.classList.toggle('hidden'));
  skipBtn.addEventListener('click',()=>{
    gameUi.animate([{opacity:1},{opacity:0}],{duration:280,fill:'forwards'});
  });

  speakerEl.textContent=dialogueData.speaker;
  textEl.textContent=dialogueData.text;
  renderChoices(dialogueData.choices);
</script>
