<script>
  function showMessage(text) {
    const message = document.getElementById('message');
    message.textContent = text;
    message.animate(
      [
        { opacity: 0, transform: 'translateY(6px)' },
        { opacity: 1, transform: 'translateY(0)' }
      ],
      { duration: 260, easing: 'ease-out' }
    );
  }
