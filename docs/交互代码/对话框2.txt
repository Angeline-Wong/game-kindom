script
   每段剧情最多配置四个选项。
  const story = [
    {
      speaker 太后,
      text 皇帝，你说什么？,
      options [
        { text 儿臣一时口误，还望母后宽恕。, reply 太后, result 既是口误，往后言行便需更加谨慎。 },
        { text 儿臣所言句句出自真心。, reply 太后, result 既是真心，那便说清楚你的打算。 },
        { text 此事关系朝局，容儿臣稍后禀明。, reply 太后, result 也罢，待朝议结束后再来见哀家。 },
        { text 儿臣还有要事，先行告退。, reply 太后, result 站住。哀家的话还没有说完。 }
      ]
    }
  ];

  const speakerEl = document.getElementById(speaker);
  const textEl = document.getElementById(dialogText);
  const optionsEl = document.getElementById(options);
  const nextBtn = document.getElementById(nextBtn);
  const dialogSystem = document.getElementById(dialogSystem);
  const skipBtn = document.getElementById(skipBtn);

  let selected = false;

  function renderOptions(items){
    optionsEl.innerHTML = ;
    items.slice(0,4).forEach((item,index)={
      const button = document.createElement(button);
      button.className = option-btn;
      button.textContent = item.text;
      button.addEventListener(click,()=chooseOption(item,index));
      optionsEl.appendChild(button);
    });
  }

  function chooseOption(item,index){
    if(selected) return;
    selected = true;

    [...optionsEl.children].forEach((btn,i)={
      btn.disabled = true;
      if(i !== index) btn.style.opacity = .48;
    });

    setTimeout(()={
      optionsEl.classList.add(hidden);
      speakerEl.textContent = item.reply;
      typeText(item.result);
      nextBtn.classList.remove(hidden);
    },180);
  }

  function typeText(text){
    textEl.textContent = ;
    let i = 0;
    const timer = setInterval(()={
      textEl.textContent += text[i++];
      if(i = text.length) clearInterval(timer);
    },38);
  }

  nextBtn.addEventListener(click,()={
    if(!selected){
      nextBtn.classList.add(hidden);
      renderOptions(story[0].options);
    }else{
      dialogSystem.classList.add(fade-out);
      setTimeout(()=dialogSystem.classList.add(hidden),240);
    }
  });

  skipBtn.addEventListener(click,()={
    dialogSystem.classList.add(fade-out);
    setTimeout(()=dialogSystem.classList.add(hidden),240);
  });

   初始状态直接展示四个选项，便于查看效果。
  nextBtn.classList.add(hidden);
  renderOptions(story[0].options);
