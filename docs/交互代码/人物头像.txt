<script>
document.querySelectorAll(".person-btn").forEach(button=>{
  const clearPressed=()=>button.classList.remove("pressed");

  button.addEventListener("pointerdown",()=>{
    button.classList.add("pressed");
  });

  button.addEventListener("pointerup",clearPressed);
  button.addEventListener("pointercancel",clearPressed);
  button.addEventListener("pointerleave",clearPressed);

  button.addEventListener("click",()=>{
    document.querySelectorAll(".person-btn").forEach(item=>{
      item.classList.remove("selected","ripple");
      item.classList.add("idle");
    });

    button.classList.remove("idle");
    button.classList.add("selected","clicked");

    void button.offsetWidth;
    button.classList.add("ripple");

    window.dispatchEvent(new CustomEvent("game-character-click",{
      detail:{
        name:button.dataset.name
      }
    }));

    setTimeout(()=>{
      button.classList.remove("ripple");
    },600);
  });
});
</script>
