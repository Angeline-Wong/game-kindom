let filter="全部";
let vacancyOnly=true;
let selectedPalace=null;
let selectedRoom=null;

const list=document.getElementById("palaceList");
const confirmBtn=document.getElementById("confirmBtn");
const searchInput=document.getElementById("searchInput");
const toast=document.getElementById("toast");

function roomAllowed(room){
  return CONSORT.rankLevel <= room.minRank;
}

function render(){
  const keyword=searchInput.value.trim();
  const rows=palaces.filter(p=>{
    const category=filter==="全部" || p.group===filter || p.level===filter;
    const vacancy=!vacancyOnly || p.rooms.some(r=>r.vacant);
    const search=!keyword || p.name.includes(keyword);
    return category && vacancy && search;
  });

  list.innerHTML="";
  if(!rows.length){
    list.innerHTML='<div class="empty">暂无符合条件的宫殿</div>';
    return;
  }

  rows.forEach(p=>{
    const anyAllowed=p.rooms.some(r=>roomAllowed(r) && (!vacancyOnly || r.vacant));
    const card=document.createElement("article");
    card.className=`palace-card ${selectedPalace===p?"selected":""} ${anyAllowed?"":"disabled"}`;

    const rooms=p.rooms.map(r=>{
      const allowed=roomAllowed(r);
      const available=r.vacant;
      const disabled=!allowed || !available;
      const reason=!available?"已有人居住":!allowed?`需${rankText(r.minRank)}以上`:"可入住";
      return `
        <button class="room-btn ${selectedPalace===p && selectedRoom===r?"selected":""}"
          data-palace="${p.name}" data-room="${r.name}" ${disabled?"disabled":""}>
          <b>${r.name}</b>
          ${reason}
        </button>
      `;
    }).join("");

    card.innerHTML=`
      <span class="check">✓</span>
      <div class="palace-head">
        <div class="palace-icon">${p.name.slice(0,1)}</div>
        <div>
          <div class="palace-name">${p.name}</div>
          <div class="palace-meta">${p.group} · ${p.level}宫殿</div>
          <div class="palace-desc">${p.desc}</div>
        </div>
        <div class="level-badge">${p.level}</div>
      </div>
      <div class="room-options">${rooms}</div>
      ${anyAllowed?"":'<div class="reason error">当前位份暂无可入住殿室</div>'}
    `;

    card.querySelectorAll(".room-btn").forEach(btn=>{
      btn.onclick=e=>{
        e.stopPropagation();
        const room=p.rooms.find(r=>r.name===btn.dataset.room);
        selectedPalace=p;
        selectedRoom=room;
        confirmBtn.disabled=false;
        render();
      };
    });

    list.appendChild(card);
  });
}

function rankText(level){
  const map={1:"皇后",2:"贵妃",3:"妃",4:"嫔",5:"贵人",6:"常在",7:"答应"};
  return map[level] || "对应位份";
}

document.querySelectorAll(".tab").forEach(btn=>{
  btn.onclick=()=>{
    document.querySelectorAll(".tab").forEach(x=>x.classList.remove("active"));
    btn.classList.add("active");
    filter=btn.dataset.filter;
    selectedPalace=null;
    selectedRoom=null;
    confirmBtn.disabled=true;
    render();
  };
});

document.getElementById("vacancyBtn").onclick=e=>{
  vacancyOnly=!vacancyOnly;
  e.currentTarget.classList.toggle("active",vacancyOnly);
  e.currentTarget.textContent=vacancyOnly?"仅看空缺":"显示全部";
  selectedPalace=null;
  selectedRoom=null;
  confirmBtn.disabled=true;
  render();
};

searchInput.oninput=render;

confirmBtn.onclick=()=>{
  if(!selectedPalace || !selectedRoom)return;

  document.getElementById("targetHome").innerHTML=
    `${selectedPalace.name}<small>${selectedRoom.name}</small>`;
  document.getElementById("confirmText").textContent=
    `徐珍淑将由翊坤宫主殿迁往${selectedPalace.name}${selectedRoom.name}，迁宫后将更新宫殿归属与居住权限。`;
  document.getElementById("favorEffect").textContent=signed(selectedRoom.favor);
  document.getElementById("powerEffect").textContent=signed(selectedRoom.power);
  document.getElementById("prestigeEffect").textContent=signed(selectedRoom.prestige);
  document.getElementById("confirmLayer").classList.add("show");
};

function signed(v){
  if(v===0)return "不变";
  return v>0?`+${v}`:`${v}`;
}

document.getElementById("backBtn").onclick=()=>document.getElementById("confirmLayer").classList.remove("show");
document.getElementById("executeBtn").onclick=()=>{
  document.getElementById("confirmLayer").classList.remove("show");

};
document.getElementById("cancelBtn").onclick=()=>showToast("已取消迁宫");
document.getElementById("closeBtn").onclick=()=>document.getElementById("panel").style.display="none";

function showToast(text){
  toast.textContent=text;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer=setTimeout(()=>toast.classList.remove("show"),1300);
}
