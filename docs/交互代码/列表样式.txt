<script>
const rankWeight = {
  "皇后": 9,
  "皇贵妃": 8,
  "贵妃": 7,
  "妃": 6,
  "嫔": 5,
  "贵人": 4,
  "常在": 3,
  "答应": 2,
  "官女子": 1
};

const qualityWeight = {
  "UR": 4,
  "SSR": 3,
  "SR": 2,
  "R": 1
};

const concubines = [
  {name:"顾明珠", rank:"皇后", palace:"坤宁宫", favor:88, charm:94, heirs:3, age:24, quality:"UR", status:["执掌六宫","凤印在手"]},
  {name:"徐珍淑", rank:"贵妃", palace:"翊坤宫", favor:92, charm:88, heirs:2, age:22, quality:"SSR", status:["圣眷正隆","可晋升"]},
  {name:"沈清漪", rank:"妃", palace:"储秀宫", favor:85, charm:93, heirs:1, age:20, quality:"SSR", status:["身怀有孕"]},
  {name:"柳如烟", rank:"妃", palace:"景仁宫", favor:78, charm:90, heirs:1, age:23, quality:"SSR", status:["擅长琴艺"]},
  {name:"苏婉宁", rank:"嫔", palace:"长春宫", favor:74, charm:86, heirs:0, age:19, quality:"SR", status:["初承圣宠"]},
  {name:"陆昭仪", rank:"嫔", palace:"咸福宫", favor:69, charm:82, heirs:1, age:25, quality:"SR", status:["可晋升"]},
  {name:"叶兰心", rank:"贵人", palace:"延禧宫", favor:63, charm:84, heirs:0, age:18, quality:"SR", status:["善解人意"]},
  {name:"白芷柔", rank:"贵人", palace:"启祥宫", favor:57, charm:81, heirs:0, age:21, quality:"R", status:["新入宫闱"]}
];

let currentRank = "全部";
let currentSort = "rank";
let currentDirection = "desc";
let selected = null;

const cards = document.getElementById("cards");
const searchInput = document.getElementById("searchInput");
const detailSheet = document.getElementById("detailSheet");
const sortSheet = document.getElementById("sortSheet");
const sortTrigger = document.getElementById("sortTrigger");
const toast = document.getElementById("toast");

function sortLabel(){
  const map = {
    rank:"位份",
    favor:"宠爱",
    heirs:"皇嗣",
    age:"年龄",
    palace:"宫殿",
    quality:"品质"
  };
  return map[currentSort];
}

function compare(a,b){
  let av;
  let bv;

  if(currentSort === "rank"){
    av = rankWeight[a.rank] || 0;
    bv = rankWeight[b.rank] || 0;
  }else if(currentSort === "quality"){
    av = qualityWeight[a.quality] || 0;
    bv = qualityWeight[b.quality] || 0;
  }else{
    av = a[currentSort];
    bv = b[currentSort];
  }

  if(typeof av === "string"){
    return currentDirection === "asc"
      ? av.localeCompare(bv,"zh-CN")
      : bv.localeCompare(av,"zh-CN");
  }

  return currentDirection === "asc" ? av-bv : bv-av;
}

function getData(){
  const keyword = searchInput.value.trim();

  return concubines
    .filter(item => currentRank === "全部" || item.rank === currentRank)
    .filter(item => !keyword || item.name.includes(keyword) || item.palace.includes(keyword))
    .sort(compare);
}

function render(){
  const data = getData();
  document.getElementById("totalCount").textContent = data.length;
  sortTrigger.textContent = `${sortLabel()} ${currentDirection === "asc" ? "↑" : "↓"}`;

  cards.innerHTML = "";

  data.forEach((item,index)=>{
    const card = document.createElement("article");
    card.className = "card" + (selected?.name === item.name ? " selected" : "");
    card.style.animationDelay = `${index * 45}ms`;

    card.innerHTML = `
      <div class="portrait" data-initial="${item.name.slice(-1)}">
        <span class="quality">${item.quality}</span>
      </div>
      <span class="rank-ribbon">${item.rank}</span>

      <div class="name-row">
        <span class="name">${item.name}</span>
        <span class="palace">${item.palace}</span>
      </div>

      <div class="line"></div>

      <div class="stats">
        <span><b>${item.favor}</b>宠爱</span>
        <span><b>${item.heirs}</b>皇嗣</span>
        <span><b>${item.age}</b>岁</span>
      </div>

      <div class="status">
        ${item.status.map((status,i)=>`
          <span class="badge ${i === 0 && item.favor >= 85 ? "hot" : ""}">
            ${status}
          </span>
        `).join("")}
      </div>
    `;

    card.onclick = ()=>{
      selected = item;
      render();

      document.getElementById("sheetTitle").textContent = `${item.name} · ${item.rank}`;
      document.getElementById("sheetSub").textContent =
        `${item.palace} · 宠爱${item.favor} · 魅力${item.charm} · 皇嗣${item.heirs}`;

      sortSheet.classList.remove("show");
      detailSheet.classList.add("show");
    };

    cards.appendChild(card);
  });
}

function showToast(text){
  toast.textContent = text;
  toast.classList.add("show");
  clearTimeout(window.toastTimer);
  window.toastTimer = setTimeout(()=>toast.classList.remove("show"),1100);
}

document.querySelectorAll(".tab").forEach(button=>{
  button.onclick = ()=>{
    document.querySelectorAll(".tab").forEach(item=>item.classList.remove("active"));
    button.classList.add("active");
    currentRank = button.dataset.rank;
    detailSheet.classList.remove("show");
    render();
  };
});

searchInput.oninput = ()=>{
  detailSheet.classList.remove("show");
  render();
};

sortTrigger.onclick = ()=>{
  detailSheet.classList.remove("show");
  sortSheet.classList.toggle("show");
};

document.querySelectorAll(".sort-option").forEach(button=>{
  button.onclick = ()=>{
    currentSort = button.dataset.sort;
